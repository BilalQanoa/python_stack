3. Display All Relationships

SELECT
    users.first_name,
    users.last_name,
    user2.first_name AS friend_first_name,
    user2.last_name AS friend_last_name
FROM users
JOIN friendships ON users.id = friendships.user_id
LEFT JOIN users AS user2 ON friendships.friend_id = user2.id;

4. Return all users who are friends with the first user

SELECT
    user2.first_name AS friend_first_name,
    user2.last_name AS friend_last_name
FROM users
JOIN friendships ON users.id = friendships.user_id
LEFT JOIN users AS user2 ON friendships.friend_id = user2.id
WHERE users.id = 1;

6. Return the count of all friendships

7. Find out who has the most friends

SELECT
    users.first_name,
    users.last_name,
    COUNT(friendships.id) AS total_friends
FROM users
JOIN friendships ON users.id = friendships.user_id
GROUP BY users.id
ORDER BY total_friends DESC
LIMIT 1;

8. Return the friends of the third user in alphabetical order

SELECT
    user2.first_name AS friend_first_name,
    user2.last_name AS friend_last_name
FROM users
JOIN friendships ON users.id = friendships.user_id
LEFT JOIN users AS user2 ON friendships.friend_id = user2.id
WHERE users.id = 3
